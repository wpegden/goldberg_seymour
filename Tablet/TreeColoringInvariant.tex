\begin{definition}
Let $T=(y_0,e_1,y_1,\ldots,e_p,y_p)$ be a tree-sequence with
distinguished uncolored edge $e_1=e_0$.  A coloring $\pi$ is
\emph{$(T,\varphi)$-invariant} if
\[
 \pi(e_i)=\varphi(e_i)\quad(2\leq i\leq p)
 \qquad\text{and}\qquad
 \overline\pi(v)=\overline\varphi(v)\quad(v\in V(T)).
\]
The edge equalities are imposed on the displayed identities of $T-e_0$.
\end{definition}
