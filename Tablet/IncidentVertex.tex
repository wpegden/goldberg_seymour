\begin{definition}
For a multigraph $G=(V,E)$, its \emph{incident-vertex type} is
\[
 V^{\circ}(G)=\{v\in V:\text{ there is an edge identity }e\in E
 \text{ incident with }v\}.
\]
An element of $V^{\circ}(G)$ retains its original vertex together with the
fact that it is incident with at least one edge.
\end{definition}
