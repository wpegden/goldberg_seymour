\begin{definition}
A normal globally maximum triple of height $p$ is \emph{balanced} if
$p=h(G)$ and its post-trunk tree edges occur in consecutive equal-colored
pairs:
\[
 \varphi(e_{2j-1})=\varphi(e_{2j})
 \quad\hbox{whenever}\quad p<2j<t(G)=|V(T)|.
\]
The inequalities refer to the paper's one-based edge labels; the Lean
indices are shifted down by one.
\end{definition}
