\begin{definition}
For a partial $k$-edge-coloring $\varphi$, a vertex set $X$, and a color
$\alpha$, define
\[
 \partial_{\varphi,\alpha}(X)
   =\{f\in\partial_G(X):\varphi(f)=\alpha\}.
\]
This is a set of boundary edge identities, so parallel edges with the same
ends are counted separately.
\end{definition}
