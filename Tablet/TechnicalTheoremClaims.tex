\begin{definition}
Let (T) be an extended Tashkinov tree with positive rung number (n),
generated by the certified series
\(
\mathcal T=\{(T_i,\varphi_{i-1},S_{i-1},F_{i-1},\Theta_{i-1}):
1\leq i\leq n+1\}.
\)
The paper-indexed technical-claim package for (T) consists of the
following six assertions.
\begin{enumerate}
\item (V(T)) is elementary with respect to (arphi_n).
\item For every proper coloring (sigma_n) leaving precisely (e_0)
uncolored and stable for ((T_{n+1},D_n,\varphi_n)), any two colors of
which at least one is missing on (T_{n+1}) are
(T_{n+1})-interchangeable under (sigma_n).
\item If (1\leq i\leq n), (Theta_i=\mathrm{PE}), and (v_i) is its
supporting vertex with first-appearance index (m(v_i)=j), then every
proper ((T_i,D_i,\varphi_i))-stable coloring (sigma_i), leaving
precisely (e_0) uncolored, is both
\[
 (T(v_i)-v_i,D_{j-1},\varphi_{j-1})\text{-stable}
 \quad\text{and}\quad
 (T_{j-1},D_{j-1},\varphi_{j-1})\text{-stable}.
\]
Moreover, if two distinct PE slots (i,h\leq n) have distinct supporting
vertices with the same first-appearance index, then
(S_i\cap S_h=\varnothing).
\item If (Theta_n=\mathrm{PE}), with supporting vertex (v_n) and
stored colors (gamma_n,\delta_n), then for every proper
((T_n,D_n,\varphi_n))-stable coloring (sigma_n) leaving precisely
(e_0) uncolored, the ((\gamma_n,\delta_n))-component through (v_n)
contains exactly the one vertex (v_n) of (T_n).
\item Let (sigma_n) be such a stable coloring, let (delta) be
defective on (T_n), and let (v) be a nonminimum vertex of
(I[\partial_{\sigma_n,\delta}(T_n)]).  For every PE supporting vertex or
SE extension vertex (v_i) with (m(v)\leq i\leq n), one has
(v\preceq_T v_i).
\item Every proper ((T_n,D_n,\varphi_n))-stable coloring leaving precisely
(e_0) uncolored is a (arphi_n\bmod T_n) coloring.
\end{enumerate}
Here (D_i) is the exact accumulated stored-color set
\noderef{TashkinovStoredColors}, (m) is
\noderef{ETTFirstAppearanceIndex}, stability on (T(v)-v) is
\noderef{StableColoringBeforeVertex}, membership in the colored boundary
incidence set is \noderef{BoundaryColorVertex}, and the other notions are
those of \noderef{ColoredVertexSetElementary},
\noderef{TreeInterchangeable}, \noderef{DefectiveColor}, and
\noderef{ModColoring}.
\end{definition}
